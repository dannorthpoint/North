function handleSubmit(event) {

  fetch(FORMSPREE_ENDPOINT, {

        document.getElementById('successMessage').classList.add('show');

    .finally(function () {
      submitBtn.disabled = false;
      submitBtn.textContent = originalBtnText;

    });

/* ===== Gallery: the lamps follow the cursor, the art drifts behind the glass ===== */
(function () {
  var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var fine = window.matchMedia('(hover: hover) and (pointer: fine)').matches;
  var media = Array.prototype.slice.call(document.querySelectorAll('.plate-media'));
  if (!media.length) return;

  // lamps track the cursor across each frame
  if (fine && !reduce) {
    media.forEach(function (el) {
      var raf = null;
      el.addEventListener('pointermove', function (ev) {
        if (raf) return;
        raf = requestAnimationFrame(function () {
          raf = null;
          var r = el.getBoundingClientRect();
          var mx = (ev.clientX - r.left) / r.width;
          el.style.setProperty('--mx', Math.min(1, Math.max(0, mx)).toFixed(3));
          el.style.setProperty('--lit', '1');
        });
      });
      el.addEventListener('pointerleave', function () {
        el.style.setProperty('--mx', '0.5');
        el.style.setProperty('--lit', '0');
      });
    });
  }

  // the image drifts slower than its frame: depth behind the glass
  if (!reduce) {
    var imgs = media.map(function (el) { return { frame: el, img: el.querySelector('img') }; })
                    .filter(function (o) { return o.img; });
    var ticking = false;
    var update = function () {
      ticking = false;
      var vh = window.innerHeight;
      imgs.forEach(function (o) {
        var r = o.frame.getBoundingClientRect();
        if (r.bottom < -200 || r.top > vh + 200) return;
        // -1..1 across the viewport, scaled to a few px of travel
        var p = ((r.top + r.height / 2) - vh / 2) / (vh / 2);
        o.img.style.setProperty('--par', (p * -12).toFixed(2));
      });
    };
    var onScroll = function () {
      if (!ticking) { ticking = true; requestAnimationFrame(update); }
    };
    window.addEventListener('scroll', onScroll, { passive: true });
    window.addEventListener('resize', onScroll, { passive: true });
    update();
  }
})();
